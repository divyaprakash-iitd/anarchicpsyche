{{/* One for a photo in a post. The original file is never re-encoded: it is the largest srcset entry and what opens when the reader taps the photo (data-zoom-src). Smaller copies are made at quality 90 (see [imaging] in hugo.toml) so the page loads fast. Params: img (page resource), alt, sizes (CSS sizes for srcset). */}} {{- $img := .img -}} {{- $alt := .alt | default "" -}} {{- $src := partial "photo-oriented.html" $img -}} {{- if eq $img.MediaType.SubType "gif" -}} {{- else -}} {{- $set := slice -}} {{- $fallback := $src -}} {{- range $w := slice 960 1600 2560 -}} {{- if lt $w $src.Width -}} {{- $r := $src.Resize (printf "%dx" $w) -}} {{- $set = $set | append (printf "%s %dw" $r.RelPermalink $w) -}} {{- if eq $w 1600 }}{{ $fallback = $r }}{{ end -}} {{- end -}} {{- end -}} {{- $set = $set | append (printf "%s %dw" $src.RelPermalink $src.Width) -}} {{- end -}}
